```java
import java.awt.GridBagConstraints;
import java.awt.GridBagLayout;
import java.awt.Insets;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;
import java.text.NumberFormat;
import javax.swing.JButton;
import javax.swing.JFormattedTextField;
import javax.swing.JFrame;
import javax.swing.JLabel;
import javax.swing.JOptionPane;


public class FormatUserInput extends JFrame implements ActionListener {
    private JButton calcButton;                   // Triggers conversion calculations
    private JLabel distLabel;                     // Label for distance input
    private JLabel feetLabel;                     // Label for feet
    private JLabel metersLabel;                   // Label for meters
    private JLabel nauticalLabel;                 // Label for nautical miles
    private JFormattedTextField feetField;        // Displays feet
    private JFormattedTextField metersField;      // Displays meters
    private JFormattedTextField nauticalField;    // Displays nautical miles
    private JFormattedTextField distField;        // Holds distance input

    /* Constructor creates GUI components and adds GUI components
       using a GridBagLayout. */
    FormatUserInput() {
        // Used to specify GUI component layout
        GridBagConstraints layoutConst = null;

        // Set frame's title
        setTitle("Miles Converter");

        // Create labels
        distLabel = new JLabel("Distance (miles):");
        feetLabel = new JLabel("Feet:");
        metersLabel = new JLabel("Meters:");
        nauticalLabel = new JLabel("Nautical Miles:");

        // Create button and add action listener
        calcButton = new JButton("Calculate");
        calcButton.addActionListener(this);

        //Need to create my own numberformat  3 decimals
        NumberFormat nf3d = NumberFormat.getNumberInstance();
        nf3d.setMaximumFractionDigits(3);   //only shows 3 decimals
        nf3d.setMinimumFractionDigits(0);  //so it doesn't force a .000


        feetField = new JFormattedTextField(nf3d);;
        feetField.setEditable(false);
        feetField.setColumns(15);


        metersField = new JFormattedTextField(nf3d);
        metersField.setEditable(false);
        metersField.setColumns(15);

        nauticalField = new JFormattedTextField(nf3d);
        nauticalField.setEditable(false);
        nauticalField.setColumns(15);

        // Create and set-up an input field for numbers (not text)
        distField = new JFormattedTextField(nf3d);
        distField.setEditable(true);
        distField.setText("0");
        distField.setColumns(15); //width

        // Use a GridBagLayout
        setLayout(new GridBagLayout());

        // Specify component's grid location
        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(10, 10, 10, 1);
        layoutConst.gridx = 0;
        layoutConst.gridy = 0;
        add(distLabel, layoutConst);

        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(10, 1, 10, 10);
        layoutConst.gridx = 1;
        layoutConst.gridy = 0;
        add(distField, layoutConst);

        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(10, 5, 10, 10);
        layoutConst.gridx = 2;
        layoutConst.gridy = 0;
        add(calcButton, layoutConst);

        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(10, 0, 1, 10);
        layoutConst.gridx = 0;
        layoutConst.gridy = 1;
        add(feetLabel, layoutConst);

        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(1, 0, 10, 10);
        layoutConst.gridx = 0;
        layoutConst.gridy = 2;
        add(feetField, layoutConst);

        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(10, 0, 1, 10);
        layoutConst.gridx = 1;
        layoutConst.gridy = 1;
        add(metersLabel, layoutConst);

        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(1, 0, 10, 10);
        layoutConst.gridx = 1;
        layoutConst.gridy = 2;
        add(metersField, layoutConst);

        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(10, 0, 1, 10);
        layoutConst.gridx = 2;
        layoutConst.gridy = 1;
        add(nauticalLabel, layoutConst);

        layoutConst = new GridBagConstraints();
        layoutConst.insets = new Insets(1, 0, 10, 10);
        layoutConst.gridx = 2;
        layoutConst.gridy = 2;
        add(nauticalField, layoutConst);
    }

    /* Method is automatically called when an event
       occurs (e.g, Enter key is pressed) */
    @Override
    public void actionPerformed(ActionEvent event) {
        double totMiles;     // Distance
        double feet;       //5280 ft in a mile
        double meter;     //1609.34m in a mile
        double nautical;  //.868976 nautical miles in a mile

        // Get value from distance field
        totMiles = ((Number) distField.getValue()).doubleValue();

        // Check if miles input is positive
        if (totMiles >= 0.0) {
            feet = totMiles * 5280.00;
            meter = totMiles * 1609.34;
            nautical = totMiles * 0.868976;

            feetField.setValue(feet); //formatted textfield use .setValue(variable) instead of .setText
            metersField.setValue(meter);
            nauticalField.setValue(nautical);

        }
        else {
            // Show failure dialog
            JOptionPane.showMessageDialog(this, "Enter a positive distance value!");
            distField.setValue(0);
        }
    }


    public static void main(String[] args) {

        FormatUserInput myFrame = new FormatUserInput();

        myFrame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        myFrame.pack();
        myFrame.setVisible(true);
    }
}
```
### Challenges
I had to learn how to use the NumberFormat class with respect to this project. After making my
own format I had to learn the difference between .setText and .setValue.
With the formatted text field  .setText ignores any number formatting while .setValue does not.
