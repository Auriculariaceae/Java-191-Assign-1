```java
import java.awt.*;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;
import javax.swing.JButton;
import javax.swing.JFrame;
import javax.swing.JLabel;
import javax.swing.JTextField;

public class SalaryCalcButtonFrame extends JFrame implements ActionListener {
    private JLabel wageLabel;     // Label for hourly salary
    private JLabel dailyLabel;
    private JLabel weeklyLabel;
    private JLabel monthlyLabel;
    private JLabel salLabel;      // Label for yearly salary
    private JLabel hoursLabel;    // Label for hours worked
    private JTextField salField;  // Displays hourly salary
    private JTextField dailyField;
    private JTextField weeklyField;
    private JTextField monthlyField;
    private JTextField wageField; // Displays yearly salary
    private JTextField hoursField;// Displays hours worked (input)
    private JButton calcButton;   // Triggers salary calculation
    private JButton resetButton;  // Resets the page to default
    /* Constructor creates GUI components and adds GUI components
       using a GridBagLayout. */
    SalaryCalcButtonFrame() {
        // Used to specify GUI component layout
        GridBagConstraints positionConst = null;

        // Set frame's title
        setTitle("Salary");
        getContentPane().setBackground(new Color(150,150,150));

        // Set hourly and yearly salary labels
        wageLabel = new JLabel("Hourly wage:");
        hoursLabel = new JLabel("Hours worked:");
        salLabel = new JLabel("Yearly salary:");
        dailyLabel = new JLabel ("Daily salary");
        weeklyLabel = new JLabel ("Weekly salary");
        monthlyLabel = new JLabel ("Monthly salary");


        wageField = new JTextField(15);
        wageField.setEditable(true);
        wageField.setText("0");

        hoursField = new JTextField(15);
        hoursField.setEditable(true);
        hoursField.setText("0");


        salField = new JTextField(15);
        salField.setEditable(false);
        salField.setBackground(Color.WHITE); //this is put in so it isn't grey and the reset works better

        dailyField = new JTextField(15);
        dailyField.setEditable(false);
        dailyField.setBackground(Color.WHITE);

        weeklyField = new JTextField(15);
        weeklyField.setEditable(false);
        weeklyField.setBackground(Color.WHITE);

        monthlyField = new JTextField(15);
        monthlyField.setEditable(false);
        monthlyField.setBackground(Color.WHITE);

        // Create a "Calculate" button
        calcButton = new JButton("Calculate");

        // Create a "Reset" button
        resetButton = new JButton("Reset");
        resetButton.addActionListener(e-> { //huge section. Converted all the .addActionListeners into a single one
            wageField.setText("");
            salField.setText("");
            hoursField.setText("");
            dailyField.setText("");
            weeklyField.setText("");
            monthlyField.setText("");

            wageField.setBackground(Color.WHITE);
            salField.setBackground(Color.WHITE);
            hoursField.setBackground(Color.WHITE);
            dailyField.setBackground(Color.WHITE);
            weeklyField.setBackground(Color.WHITE);
            monthlyField.setBackground(Color.WHITE);
            });
        // Use "this" class to handle button presses
        calcButton.addActionListener(this);  //using "this" calls the actionPerformed(ActionEvent event) method

        // Use a GridBagLayout
        setLayout(new GridBagLayout());
        positionConst = new GridBagConstraints();

        // Specify component's grid location
        positionConst.gridx = 0;
        positionConst.gridy = 0;

        // 10 pixels of padding around component
        positionConst.insets = new Insets(10, 10, 10, 10);

        // Add component using the specified constraints
        add(wageLabel, positionConst);

        positionConst.gridx = 1;
        positionConst.gridy = 0;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(wageField, positionConst);

        positionConst.gridx = 0;
        positionConst.gridy = 1;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(hoursLabel, positionConst);

        positionConst.gridx = 1;
        positionConst.gridy = 1;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(hoursField, positionConst);

        positionConst.gridx = 0;
        positionConst.gridy = 2;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(dailyLabel, positionConst);

        positionConst.gridx = 1;
        positionConst.gridy = 2;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(dailyField, positionConst);

        positionConst.gridx = 0;
        positionConst.gridy = 3;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(weeklyLabel, positionConst);

        positionConst.gridx = 1;
        positionConst.gridy = 3;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(weeklyField, positionConst);

        positionConst.gridx = 0;
        positionConst.gridy = 4;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(monthlyLabel, positionConst);

        positionConst.gridx = 1;
        positionConst.gridy = 4;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(monthlyField, positionConst);

        positionConst.gridx = 0;
        positionConst.gridy = 5;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(salLabel, positionConst);

        positionConst.gridx = 1;
        positionConst.gridy = 5;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(salField, positionConst);

        positionConst.gridx = 0;
        positionConst.gridy = 6;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(calcButton, positionConst);

        positionConst.gridx = 1;
        positionConst.gridy = 6;
        positionConst.insets = new Insets(10, 10, 10, 10);
        add(resetButton, positionConst);
    }

    /* Method is automatically called when an event
       occurs (e.g, button is pressed) */
    @Override
    public void actionPerformed(ActionEvent event) {
        String hourlyWage;      // User specified hourly wage
        String hoursWorked;        // User entered hours worked

        // Get user's wage input
        hourlyWage = wageField.getText();
        hoursWorked = hoursField.getText();
        // Convert from String to an integer and a new variable for easier maths
        int hourWage = Integer.parseInt((hourlyWage));
        int hourWork = Integer.parseInt(hoursWorked);
        // Display calculated salary

        String yearlyWage = String.valueOf(hourWage * hourWork * 365);
        salField.setText(yearlyWage);
        salField.setBackground(new Color(215,2,222));
        salField.setForeground(Color.WHITE);

        String dailyWage = String.valueOf(hourWage * hourWork);
        dailyField.setText(dailyWage);
        dailyField.setBackground(new Color(220,240,0));
        dailyField.setForeground(Color.WHITE);

        String weeklyWage = String.valueOf(hourWage * hourWork*7);
        weeklyField.setText(weeklyWage);
        weeklyField.setBackground(new Color(1,181,26));
        weeklyField.setForeground(Color.WHITE);

        String monthlyWage = String.valueOf(hourWage * hourWork*28);
        monthlyField.setText(monthlyWage);
        monthlyField.setBackground(new Color(15,2,196));
        monthlyField.setForeground(Color.WHITE);

        //just for colors sake
        wageField.setBackground(new Color(255,1,1));
        hoursField.setBackground(new Color(240,144,0));
        wageField.setForeground(Color.WHITE);
        hoursField.setForeground(Color.WHITE);

    }

    /* Creates a SalaryCalculatorFrame and makes it visible */
    public static void main(String[] args) {
        // Creates SalaryLabelFrame and its components
        SalaryCalcButtonFrame myFrame = new SalaryCalcButtonFrame();

        myFrame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        myFrame.pack();
        myFrame.setVisible(true);
    }
}
```
